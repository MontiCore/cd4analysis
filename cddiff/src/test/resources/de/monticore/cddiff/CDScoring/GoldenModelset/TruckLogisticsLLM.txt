classdiagram TruckLogisticsLLM {

  class Location {
    String address;
  }

  class Driver {
    String firstName;
    String lastName;
  }

  class Vehicle {
    String vehicleNumber;
  }

  class Truck extends Vehicle {
    double weight;
  }

  class Car extends Vehicle {
    int numberOfSeats;
  }

  class Van extends Vehicle {}

  class Order {
    String origin;
    String destination;
    Date dateOfExecution;
  }

  // Associations

  association [1] Location -> (drivers) Driver [*];
  association [1] Location -> (vehicles) Vehicle [*];

  // Each driver can have at most one current order
  association [0..1] Order -> (assignedDrivers) Driver [1]; // at least one driver per order

  // NOTE: Drivers carry out at most one order at a time
  association [1] Driver -> (currentOrder) Order [0..1];

  // Vehicles are assigned to at most one driver
  association [0..1] Vehicle -> (assignedDriver) Driver [1];
}
