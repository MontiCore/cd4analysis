classdiagram TransportCompanyLLM {

  // ----------------- Employees -----------------

  class Employee {
    String name;
    String socialSecurityNumber;
  }

  class Driver extends Employee {}

  class Planner extends Employee {
    String username;
    String password;
  }

  class AdministrativeStaff extends Employee {
    String username;
    String password;
    String fieldOfActivity;
  }

  // ----------------- Customers -----------------

  class Customer {
    String name;
    String billingAddress;
  }

  // ----------------- Partner Company -----------------

  class PartnerCompany {
    String name;
    String address;
  }

  // ----------------- Vehicles -----------------

  class Vehicle {
    String licensePlateNumber;
    double mileage;
  }

  class RefrigeratedTruck extends Vehicle {}
  class SmallVan extends Vehicle {}
  class LoadingPlatform extends Vehicle {}
  class BoxTruck extends Vehicle {}

  // ----------------- Repair Shops & Repairs -----------------

  class RepairShop {
    String name;
    String address;
  }

  class Repair {
    double cost;
    String duration;
  }

  association [1] Repair -> (repairShop) RepairShop [1];
  association [1] Repair -> (vehicle) Vehicle [1];
  association [1] Repair -> (responsibleEmployee) Employee [1];

  // ----------------- Orders -----------------

  class Order {
    String packageSize;
    String pickupAddress;
    String deliveryAddress;
    String comment;
    DateTime startTime;
    DateTime endTime;
  }

  // Each order has exactly one customer
  association [1] Order -> (client) Customer [1]; // always exactly one client

  // Each order is placed either by Admin Staff OR PartnerCompany
  association [0..1] Order -> (placedByAdmin) AdministrativeStaff [1];
  association [0..1] Order -> (placedByPartner) PartnerCompany [1];

  // Orders involve assignment by a Planner
  association [1] Order -> (assignedBy) Planner [1];

  // Orders are carried out by a Driver with a Vehicle
  association [1] Order -> (driver) Driver [1];
  association [1] Order -> (vehicle) Vehicle [1];
}
