classdiagram GameAreaLLM {

  class GameArea {
    String name;
  }

  class GameElement {
    long x;
    long y;
  }

  class Shape extends GameElement {}

  class Object extends GameElement {}

  class Opponent extends GameElement {
    int lives;
  }

  // Associations
  association [1] GameArea -> (elements) GameElement [1..*];

  // A game area consists of one or two game pieces
  association [1] GameArea -> (pieces) GamePiece [1..2];

  class GamePiece {}

  // Shapes can be connected to any number of other shapes
  association Shape <-> (connectedShapes) Shape [*];

  // Exactly one opponent is known to the game area as the boss
  association [1] GameArea -> (boss) Opponent [1];
}
