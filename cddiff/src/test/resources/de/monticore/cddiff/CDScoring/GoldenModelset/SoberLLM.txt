classdiagram SoberLLM {

  class App {
    String appName = "SoberApp";
  }

  // ---------------- Users & Customers ----------------
  class Customer {
    String name;
    int customerNumber;
  }

  class CarOwner extends Customer {}

  // ---------------- Vehicles ----------------
  class Car {
    String licensePlate;
    boolean isSelfDriving;
  }

  // ---------------- Taxi Services ----------------
  abstract class TaxiService {
    DateTime pickupTime;
    DateTime dropoffTime;
    String pickupLocation;
    String dropoffLocation;
    double duration;
    double distance;
    int passengers;
  }

  class RideHail extends TaxiService {
    double fee;
    RideHailRequestType requestType;
  }

  class RideShare extends TaxiService {
    double negotiatedFee;
  }

  enum RideHailRequestType {
    APP,
    HAND_WAVE;
  }

  // ---------------- Incentives ----------------
  class Incentive {
    int ridesUntilTree = 20;
    String description = "Plant a tree after 20 ride-share rides.";
  }

  // ---------------- Accidents ----------------
  class Accident {
    DateTime date;
    String location;
    double damageAmount;
  }

  // ---------------- Associations ----------------
  association [1] App -> (customers) Customer [*];
  association [*] CarOwner -> (ownedCars) Car [*];
  association [*] Car -> (services) TaxiService [*];
  association [*] Customer -> (services) TaxiService [*];
  association [*] Car -> (accidents) Accident [*];
  association [*] Customer -> (incentives) Incentive [*];
}
