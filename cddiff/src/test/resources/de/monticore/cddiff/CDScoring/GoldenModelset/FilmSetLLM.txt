/* (c) CD4A Model for Film Set System */

classdiagram FilmSetLLM {

  class Employee {
    String name;
    String socialSecurityNumber;
  }

  class CreativeEmployee extends Employee {}

  class Actor extends CreativeEmployee {
    String education;
    boolean hasOscar;
  }

  class Director extends CreativeEmployee {
    String assistantName;
    boolean hasOscar;
  }

  class Technician extends Employee {
    String fieldOfActivity;
  }

  class FilmSet {
    String location;
  }

  class Film {
    String title;
    int releaseYear;
    Genre genre;
  }

  enum Genre {
    THRILLER,
    ACTION,
    HORROR,
    COMEDY;
  }

  class Screenwriter {
    String name;
    String mostSuccessfulFilm;
  }

  class Screenplay {
    String title;
    String version;
    Date date;
    int numberOfScenes;
    String plot;
  }

  class Novel {
    String title;
    String author;
  }

  class Concept {
    String title;
  }

  // Associations
  association [1] FilmSet -> Employee [*];
  association [*] FilmSet <-> Film [*];
  association [1] Screenwriter -> Screenplay [*];
  association [1] Screenplay -> (film) Film [1];
  association [1] Film -> (screenplay) Screenplay [1];

  // A Screenplay is based either on a Novel or on a Concept
  association [0..1] Screenplay -> (basedOnNovel) Novel [1];
  association [0..1] Screenplay -> (basedOnConcept) Concept [1];

  // Actors prepare for their roles by reading the screenplay
  association [*] Actor -> (reads) Screenplay [*];

  // Directors implement the screenplay
  association [*] Director -> Screenplay [*];
}
