/* (c) CD4A Model based on given project description */

classdiagram BuildingManagementLLM {

  class WebPortal {
    String version;
    String url;
  }

  class Building {
    String buildingId;
    String name;
    EnergyClass energyClass;
    String address;
    int yearOfConstruction;
  }

  enum EnergyClass {
    A,
    B,
    C,
    D;
  }

  class AdditionalEntry {
    String name;
    boolean isPublic;
  }

  class TextEntry extends AdditionalEntry {
    String text;
  }

  class NumericEntry extends AdditionalEntry {
    double value;
    String unit;
  }

  class Image {
    String filename;
    boolean isProfile;
  }

  class User {
    String username;
    String email;
    String password;
  }

  class Administrator extends User {
    String idCardNumber;
  }

  class Comment {
    String text;
  }

  // ---------------- Associations ----------------

  association [1] WebPortal -> (catalog) Building [*];
  association [1] Building -> (owner) User [1];
  association [1] Building -> (entries) AdditionalEntry [*];
  association [1] Building -> (images) Image [*];
  association [1] Building -> (comments) Comment [*];
  association [1] Comment -> (author) User [1];

}
