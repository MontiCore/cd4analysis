classdiagram SchoolLLM {

  class School {
    String name;
  }

  class Teacher {
    String firstName;
    String lastName;
    String email;
  }

  class Student {
    String firstName;
    String lastName;
  }

  class ClassGroup {
    String name;
  }

  class Room {
    String name;
  }

  class Classroom extends Room {
    int capacity;
  }

  class Gym extends Room {
    double sizeSquareMeters;
  }

  // Associations

  association [1] Student -> (school) School [1];
  association [*] Teacher -> (teachesAt) School [0..3];

  association [1] School -> (principal) Teacher [1];
  // NOTE: Each teacher can only be principal of one school – not enforceable in CD4A.

  association [*] School -> (rooms) Room [*];

  association [10..31] Student -> (classGroup) ClassGroup [1];
  association [1] ClassGroup -> (classroom) Classroom [1];

}
