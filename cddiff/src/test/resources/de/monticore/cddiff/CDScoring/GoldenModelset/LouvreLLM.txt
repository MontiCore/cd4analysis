classdiagram LouvreMuseum {

  // ---------------- Core Museum Structure ----------------
  class Museum {
    String name;
    String address;
  }

  class Location {
    String name;
    String description;
  }

  association [1] Museum -> (locations) Location [*];

  class Exhibition {
    String title;
    Date startDate;
    Date endDate;
    Date planningStartDate;
  }

  association [1] Location -> (exhibitions) Exhibition [*];

  // ---------------- Employees ----------------
  class Employee {
    String name;
    String role;
  }

  association coordinator [1] Exhibition -> (coordinator) Employee [1];

  // Coaching relation: one senior can coach multiple juniors, each junior has one senior
  association coaching [*] Employee (junior) -> (senior) Employee [1];

  // ---------------- Exhibition Items ----------------
  class ExhibitionItem {
    String description;
  }

  association [1] Exhibition -> (desiredItems) ExhibitionItem [*];

  class Piece {
    String title;
    String artist;
    String period;
  }

  class Collector {
    String name;
    String contactInfo;
  }

  // Some items may be linked to several possible pieces
  association [1] ExhibitionItem -> (candidatePieces) Piece [*];

  // Requested pieces are tracked per collector
  association [*] Collector -> (pieces) Piece [*];
}
