/* (c) EBike System Model */

classdiagram EBikeLLM {

  class EBike {}

  class Frame {
    String material = "Steel";
  }

  class Wheel {}

  class DriveSystem {}

  class Motor {}

  class Battery {
    double storedEnergyWh;
  }

  class Controller {}

  class BasicController extends Controller {}

  class AdvancedController extends Controller {
    Date nextMaintenanceDate;
  }

  enum ControllerState {
    ON,
    OFF,
    CHARGING;
  }

  // Associations
  association [1] EBike -> (frame) Frame [1];
  association [1] EBike -> (driveSystem) DriveSystem [1];
  association [1] EBike -> (controller) Controller [1];
  association [1] Frame -> (wheels) Wheel [2];

  association [1] DriveSystem -> (motor) Motor [1];

  // "Removable battery" means optional:
  association [1] EBike -> (battery) Battery [0..1];

  // Controller manages battery if connected:
  association [0..1] Controller -> (controlledBattery) Battery [0..1];

  // Controller commands the drive system:
  association [1] Controller -> (controlledDrive) DriveSystem [1];

}
